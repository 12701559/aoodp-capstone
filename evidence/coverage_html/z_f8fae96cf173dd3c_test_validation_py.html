29class TestAmount: 

30 @pytest.mark.parametrize("value", [0.01, 1, 1250.0, MAX_AMOUNT]) 

31 def test_accepts_valid_amounts(self, value: float) -> None: 

32 assert validate_amount(value) == float(value) 

33 

34 @pytest.mark.parametrize( 

35 "value", 

36 [ 

37 0, # zero is not a transaction 

38 -1.0, # negative 

39 MAX_AMOUNT * 2, # above the sanity ceiling 

40 float("nan"), # NaN passes every comparison 

41 float("inf"), # inf breaks every threshold rule 

42 "100", # a string is not a number 

43 None, 

44 True, # bool is not an amount, despite subclassing int 

45 ], 

46 ) 

47 def test_rejects_invalid_amounts(self, value: object) -> None: 

48 with pytest.raises(ValidationError): 

49 validate_amount(value) 

50 

51 def test_zero_permitted_only_when_explicitly_allowed(self) -> None: 

52 assert validate_amount(0, allow_zero=True) == 0.0

138class TestRedaction: 

139 def test_account_number_keeps_only_the_suffix(self) -> None: 

140 assert redact_account_number("ACC-00000001") == "********0001" 

141 assert redact_account_number("1234") == "****" 

142 assert redact_account_number("12") == "****" 

143 

144 def test_redacted_mapping_is_a_copy_and_masks_sensitive_keys(self) -> None: 

145 original = {"account_number": "ACC-00000001", "amount": 12.5} 

146 redacted = redact_mapping(original) 

147 

148 assert redacted is not original 

149 assert redacted["account_number"] == "********0001" 

150 assert redacted["amount"] == 12.5 

151 # The caller's object is untouched: a redacted view must never 

152 # become the object later read by scoring logic. 

153 assert original["account_number"] == "ACC-00000001" 

154 

155 @pytest.mark.parametrize( 

156 "key", ["customer_id", "pan", "iban_ref", "email_address", "phone"] 

157 ) 

158 def test_each_sensitive_marker_is_masked(self, key: str) -> None: 

159 # A nine-character value keeps only its last four characters visible, 

160 # so the marker matching is what is being asserted here, not the 

161 # length of the mask. 

162 masked = redact_mapping({key: "sensitive"})[key] 

163 assert masked == "*****tive" 

164 assert masked != "sensitive" 

165 

166 def test_extra_sensitive_keys_can_be_declared_by_the_caller(self) -> None: 

167 redacted = redact_mapping({"nickname": "Jules"}, extra_sensitive=("nickname",)) 

168 assert redacted["nickname"] == "*ules" 

169
